function capturarGeolocalizacao() {
  const btn    = document.getElementById('btnGeo');
  const status = document.getElementById('geoStatus');
  const latEl  = document.getElementById('geoLat');
  const lngEl  = document.getElementById('geoLng');
  if (!navigator.geolocation) {
    status.textContent = '❌ GPS não suportado neste dispositivo.';
    status.style.color = '#e53e3e'; return;
  }
  btn.disabled = true;
  btn.textContent = '📡 Buscando...';
  navigator.geolocation.getCurrentPosition(
    pos => {
      const lat = pos.coords.latitude.toFixed(6);
      const lng = pos.coords.longitude.toFixed(6);
      latEl.value = lat; lngEl.value = lng;
      btn.textContent = '✅ GPS capturado';
      btn.style.background = '#38a169';
      btn.disabled = false;
      status.innerHTML = `<strong>${lat}, ${lng}</strong> &nbsp;<a href="https://maps.google.com/?q=${lat},${lng}" target="_blank" style="color:#3182ce;">🗺 Ver no mapa</a>`;
      status.style.color = '#22543d';
    },
    err => {
      const m = {1:'Permissão negada — libere o GPS.',2:'GPS indisponível.',3:'Tempo esgotado.'};
      btn.textContent = '📍 Capturar Localização GPS';
      btn.style.background = '#3182ce';
      btn.disabled = false;
      status.textContent = '⚠️ ' + (m[err.code]||'Erro.');
      status.style.color = '#e53e3e';
    },
    {enableHighAccuracy:true, timeout:12000, maximumAge:0}
  );
}
